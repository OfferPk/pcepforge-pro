/\*\*  
 \* PCEPForge Pro \- Multi-Vendor Configuration Synthesizer  
 \* Generates verified production syntax for Cisco IOS-XR, Arista EOS,  
 \* Juniper Junos, Nokia SR-OS, and Prometheus Alerting Rules.  
 \*/

(function (root, factory) {  
  if (typeof module \=== 'object' && module.exports) {  
    module.exports \= factory();  
  } else {  
    root.PCEPConfigSynthesizer \= factory();  
  }  
})(typeof self \!== 'undefined' ? self : this, function () {

  /\*\*  
   \* Generates Cisco IOS-XR PCEP & SR-TE PCC Configuration  
   \*/  
  function generateCiscoIOSXR(cfg) {  
    const lines \= \[\];  
    lines.push('\! \=======================================================');  
    lines.push('\! Cisco IOS-XR Segment Routing Traffic Engineering (SR-TE)');  
    lines.push('\! Path Computation Client (PCC) Hardened Configuration');  
    lines.push(\`\! Generated by PCEPForge Pro for Node: \${cfg.hostname || 'NYC-CR-01'}\`);  
    lines.push('\! \=======================================================');  
    lines.push('');

    const pPce \= cfg.primaryPceIp || '198.51.100.10';  
    const sPce \= cfg.secondaryPceIp || '198.51.100.11';  
    const ka \= cfg.keepaliveSec || 30;  
    const dt \= cfg.deadTimerSec || 120;

    lines.push('segment-routing');  
    lines.push(' traffic-eng');  
    lines.push('  pcc');  
    lines.push(\`   source-address ipv4 \${cfg.localIp || '198.51.100.1'}\`);  
    lines.push(\`   pce address ipv4 \${pPce}\`);  
    lines.push('    precedence 10');  
    if (cfg.authType \=== 'MD5' && cfg.md5Key) {  
      lines.push(\`    password clear \${cfg.md5Key}\`);  
    } else if (cfg.authType \=== 'TLS') {  
      lines.push('    tls');  
    }  
    lines.push('   \!');  
    if (cfg.secondaryPceIp) {  
      lines.push(\`   pce address ipv4 \${sPce}\`);  
      lines.push('    precedence 20');  
      if (cfg.authType \=== 'MD5' && cfg.md5Key) {  
        lines.push(\`    password clear \${cfg.md5Key}\`);  
      } else if (cfg.authType \=== 'TLS') {  
        lines.push('    tls');  
      }  
      lines.push('   \!');  
    }  
    lines.push('   timers');  
    lines.push(\`    keepalive \${ka}\`);  
    lines.push(\`    dead-timer \${dt}\`);  
    if (cfg.delegated) {  
      lines.push('    delegation-timeout 60');  
    }  
    lines.push('   \!');  
    if (cfg.dbVersioningEnabled) {  
      lines.push('   stateful-client');  
      lines.push('    instantiation');  
      lines.push('    redundancy');  
      lines.push('   \!');  
    }  
    lines.push('  \!');  
    lines.push('');  
    lines.push('  \! Segment Routing Policy (PCE-Delegated)');  
    lines.push('  policy POL-SRTE-NYC-LON');  
    lines.push('   color 100 end-point ipv4 198.51.100.2');  
    lines.push('   candidate-paths');  
    lines.push('    preference 100');  
    lines.push('     dynamic');  
    lines.push('      pce');  
    if (cfg.latencyConstrained) {  
      lines.push('       metric');  
      lines.push('        type latency');  
      lines.push('       \!');  
    }  
    lines.push('      \!');  
    lines.push('     \!');  
    lines.push('    \!');  
    lines.push('   \!');  
    lines.push('  \!');  
    lines.push(' \!');  
    lines.push('\!');

    return lines.join('\\n');  
  }

  /\*\*  
   \* Generates Arista EOS PCEP Configuration  
   \*/  
  function generateAristaEOS(cfg) {  
    const lines \= \[\];  
    lines.push('\! \=======================================================');  
    lines.push('\! Arista EOS Segment Routing PCEP Configuration');  
    lines.push(\`\! Generated by PCEPForge Pro for Node: \${cfg.hostname || 'SJC-SPINE-01'}\`);  
    lines.push('\! \=======================================================');  
    lines.push('');

    const pPce \= cfg.primaryPceIp || '10.100.1.10';  
    const ka \= cfg.keepaliveSec || 30;  
    const dt \= cfg.deadTimerSec || 120;

    lines.push('router traffic-engineering');  
    lines.push(' pcep');  
    lines.push(\`  local-address \${cfg.localIp || '10.100.1.1'}\`);  
    lines.push(\`  pce primary \${pPce}\`);  
    if (cfg.authType \=== 'MD5' && cfg.md5Key) {  
      lines.push(\`   authentication md5 \${cfg.md5Key}\`);  
    }  
    if (cfg.secondaryPceIp) {  
      lines.push(\`  pce secondary \${cfg.secondaryPceIp}\`);  
    }  
    lines.push(\`  timers keepalive \${ka} dead-timer \${dt}\`);  
    if (cfg.delegated) {  
      lines.push('  stateful delegation enable');  
      lines.push('  pce-initiated enable');  
    }  
    lines.push('  no shutdown');  
    lines.push(' \!');  
    lines.push('\!');

    return lines.join('\\n');  
  }

  /\*\*  
   \* Generates Juniper Junos PCEP Configuration  
   \*/  
  function generateJuniperJunos(cfg) {  
    const lines \= \[\];  
    lines.push('\# \=======================================================');  
    lines.push('\# Juniper Junos Security & PCEP Protocol Configuration');  
    lines.push(\`\# Generated by PCEPForge Pro for Node: \${cfg.hostname || 'LON-CORE-01'}\`);  
    lines.push('\# \=======================================================');  
    lines.push('');

    const pPce \= cfg.primaryPceIp || '198.51.100.10';  
    const ka \= cfg.keepaliveSec || 30;  
    const dt \= cfg.deadTimerSec || 120;

    lines.push(\`set protocols pcep pce PCE-PRIMARY address \${pPce}\`);  
    lines.push(\`set protocols pcep pce PCE-PRIMARY destination-ipv4-address \${pPce}\`);  
    lines.push(\`set protocols pcep pce PCE-PRIMARY local-ipv4-address \${cfg.localIp || '198.51.100.1'}\`);  
    if (cfg.authType \=== 'MD5' && cfg.md5Key) {  
      lines.push(\`set protocols pcep pce PCE-PRIMARY authentication-key "\${cfg.md5Key}"\`);  
    } else if (cfg.authType \=== 'TLS') {  
      lines.push(\`set protocols pcep pce PCE-PRIMARY tls-profile PCEP-TLS-PROFILE\`);  
    }  
    lines.push(\`set protocols pcep pce PCE-PRIMARY timers keepalive \${ka}\`);  
    lines.push(\`set protocols pcep pce PCE-PRIMARY timers dead-timer \${dt}\`);  
    if (cfg.delegated) {  
      lines.push(\`set protocols pcep pce PCE-PRIMARY pce-type stateful\`);  
      lines.push(\`set protocols pcep pce PCE-PRIMARY lsp-provisioning\`);  
    }

    if (cfg.secondaryPceIp) {  
      lines.push(\`set protocols pcep pce PCE-STANDBY address \${cfg.secondaryPceIp}\`);  
      lines.push(\`set protocols pcep pce PCE-STANDBY local-ipv4-address \${cfg.localIp || '198.51.100.1'}\`);  
      lines.push(\`set protocols pcep pce PCE-STANDBY pce-type stateful\`);  
    }

    lines.push('');  
    lines.push('\# LSP Delegation to PCE');  
    lines.push('set protocols mpls label-switched-path LSP-TO-NYC to 198.51.100.2');  
    lines.push('set protocols mpls label-switched-path LSP-TO-NYC lsp-flavor segment-routing');  
    if (cfg.delegated) {  
      lines.push('set protocols mpls label-switched-path LSP-TO-NYC delegate');  
    }

    return lines.join('\\n');  
  }

  /\*\*  
   \* Generates Nokia SR-OS PCEP Configuration  
   \*/  
  function generateNokiaSROS(cfg) {  
    const lines \= \[\];  
    lines.push('\# \=======================================================');  
    lines.push('\# Nokia SR-OS PCEP / Stateful PCC Configuration');  
    lines.push(\`\# Generated by PCEPForge Pro for Node: \${cfg.hostname || 'NOKIA-PE-01'}\`);  
    lines.push('\# \=======================================================');  
    lines.push('');

    const pPce \= cfg.primaryPceIp || '198.51.100.10';

    lines.push('configure router pcep');  
    lines.push(\`    pcc-address \${cfg.localIp || '198.51.100.1'}\`);  
    lines.push(\`    pce "\${pPce}"\`);  
    lines.push(\`        address \${pPce}\`);  
    if (cfg.authType \=== 'MD5' && cfg.md5Key) {  
      lines.push(\`        authentication-key "\${cfg.md5Key}" hash2\`);  
    }  
    lines.push('        no shutdown');  
    lines.push('    exit');  
    if (cfg.delegated) {  
      lines.push('    stateful-pce');  
      lines.push('        enable');  
      lines.push('        pce-initiated-lsp enable');  
      lines.push('    exit');  
    }  
    lines.push('    no shutdown');  
    lines.push('exit');

    return lines.join('\\n');  
  }

  /\*\*  
   \* Generates Prometheus PCEP Telemetry & Alertmanager Rules  
   \*/  
  function generatePrometheusRules(cfg) {  
    const lines \= \[\];  
    lines.push('\# \=======================================================');  
    lines.push('\# Prometheus & OpenTelemetry PCEP Alerting Rules');  
    lines.push('\# Standards: RFC 5440 / RFC 8231');  
    lines.push('\# \=======================================================');  
    lines.push('');  
    lines.push('groups:');  
    lines.push('  \- name: pcep\_traffic\_engineering\_alerts');  
    lines.push('    rules:');  
    lines.push('      \- alert: PCEPSessionDown');  
    lines.push('        expr: pcep\_session\_state \!= 1');  
    lines.push('        for: 15s');  
    lines.push('        labels:');  
    lines.push('          severity: critical');  
    lines.push('        annotations:');  
    lines.push('          summary: "PCEP session down to controller {{ \$labels.pce\_address }}"');  
    lines.push('          description: "PCC has lost connection to primary Path Computation Element. Traffic steering is frozen."');  
    lines.push('');  
    lines.push('      \- alert: PCEPErrorRateSpike');  
    lines.push('        expr: rate(pcep\_messages\_pcerr\_total\[1m\]) \> 2');  
    lines.push('        for: 30s');  
    lines.push('        labels:');  
    lines.push('          severity: warning');  
    lines.push('        annotations:');  
    lines.push('          summary: "PCEP Error (PCErr) rate elevated"');  
    lines.push('          description: "Controller rejecting PCReq/PCRpt messages (e.g. unknown SID, MSD exceeded, or parsing error)."');  
    lines.push('');  
    lines.push('      \- alert: PCEPDelegationRevocation');  
    lines.push('        expr: increase(pcep\_lsp\_delegation\_revocations\_total\[5m\]) \> 0');  
    lines.push('        labels:');  
    lines.push('          severity: critical');  
    lines.push('        annotations:');  
    lines.push('          summary: "LSP delegation revoked on {{ \$labels.instance }}"');  
    lines.push('          description: "Router revoked LSP delegation from PCE due to timeout or synchronization mismatch."');

    return lines.join('\\n');  
  }

  return {  
    generateCiscoIOSXR: generateCiscoIOSXR,  
    generateAristaEOS: generateAristaEOS,  
    generateJuniperJunos: generateJuniperJunos,  
    generateNokiaSROS: generateNokiaSROS,  
    generatePrometheusRules: generatePrometheusRules  
  };  
});
